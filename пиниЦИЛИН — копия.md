## 2. Скрипт — `lab3.js`

```javascript
// ============ Лабораторная работа №3 ============
// Итерируемые конструкции JavaScript

// ---- 3. Расширяем стандартный класс Object ----
class IterableObject extends Object {
  constructor(object) {
    super();
    // копируем свойства переданного объекта в this
    Object.assign(this, object);
  }

  // Метод Symbol.iterator — именно по нему for...of понимает,
  // что объект можно итерировать. Должен возвращать итератор.
  [Symbol.iterator]() {
    // получаем пары [ключ, значение] объекта в константу
    const entries = Object.entries(this);
    let index = -1;

    return {
      // ---- 4. Метод next ----
      // на каждом витке возвращает {value, done},
      // где value — объект вида {value: "значение", key: "название ячейки"}
      next() {
        index++;

        // перебор закончен
        if (index >= entries.length) {
          return { value: undefined, done: true };
        }

        return {
          value: {
            value: entries[index][1], // значение ячейки
            key:   entries[index][0]  // название ключа
          },
          done: false
        };
      }
    };
  }
}

// ---- 2. Массив данных (news подключается из data.js, лаб. №2) ----
// Берём один объект из массива — одну новость
const sourceObject = news[0];

// ---- 3. Создаём итерируемый объект нового типа ----
const iterableObject = new IterableObject(sourceObject);

console.log('Исходный объект (новость):');
console.log(sourceObject);

// ---- 4. Проверяем, что возвращает Symbol.iterator на каждом витке ----
console.log('\n--- Ручные вызовы итератора ---');
const iterator = iterableObject[Symbol.iterator]();

console.log(iterator.next());
console.log(iterator.next());

// ---- 5. Итерация объекта нового типа в цикле for ... of ----
console.log('\n--- Итерация через for ... of ---');

for (let element of iterableObject) {
  // element — это объект {value: "значение", key: "название ячейки"}
  console.log(element);
}

// Проитерированный объект
console.log('\nПроитерированный объект:');
console.log(iterableObject);

// Дополнительно: простой объект из теоретической части
console.log('\n--- Простой объект (как в теории) ---');
const simpleObject = new IterableObject({ 1: 'a', 2: 'b', 3: 'c' });

for (let element of simpleObject) {
  console.log(element);
}
```

## Как это работает

- `[Symbol.iterator]()` при каждом вызове создаёт **новый** итератор со своим счётчиком `index`. Поэтому после ручных вызовов `next()` цикл `for...of` всё равно начинает перебор с начала — он вызывает `[Symbol.iterator]()` заново.
- `next()` возвращает стандартную структуру `{value, done}`, но `value` — не просто значение, а объект `{value: ..., key: ...}`, как требует задание.
- Ручные вызовы `iterator.next()` показывают в консоли полную структуру ответа итератора (`{value: {...}, done: false}`), а цикл `for...of` — только значение каждого витка.
